# Chiron Stick GRUB theme (setup/50-rice.sh installs it to /boot/grub/themes/chiron)

# The picture is 16:9. Scaled to the screen's height and centred, the logo stays round on 4:3 and
# 16:10 modes too (the sky is cropped at the sides); stretching squashed it
desktop-image-scale-method: "fitheight"
desktop-image-h-align: "center"
desktop-image-v-align: "center"
desktop-color: "#0b0a12"
title-text: ""
message-font: "DejaVu Sans Regular 16"
message-color: "#ece9f4"
terminal-font: "DejaVu Sans Mono Regular 14"
# The console (shown while the kernel loads, and for the 'c' command line) gets a thin violet edge.
# GRUB doesn't place it where these lines ask: with them it centres an 80x24 console, which is still
# smaller than its default box covering 70% of the screen. Inside it stays black.
terminal-box: "terminal_box_*.png"
terminal-left: "15%"
terminal-top: "50%"
terminal-width: "70%"
terminal-height: "36%"
terminal-border: "0"

# Sized for 640x480 too (legacy BIOS often runs GRUB at that size): the longest entry,

    item_padding = 10
    item_spacing = 4
    scrollbar = false
}

# With Secure Boot GRUB may not load font files (shim_lock policy), so its own text is in the built-in
# pixel font. The countdown is a thin bar instead (it fills, then the stick starts), and the key hint
# is drawn smooth into background.png just below it (make-assets.py). GRUB makes a progress bar at
# least 28 px tall, so the line is drawn by 28 px pictures that are clear above and below it.
+ progress_bar {
    id = "__timeout__"
    left = 40%
    width = 20%
    top = 87%
    height = 28
    show_text = false
    bar_style = "track_*.png"
    highlight_style = "fill_*.png"
}
